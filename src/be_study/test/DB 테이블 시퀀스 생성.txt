	CREATE TABLE Vilage_Fcst (
	    id NUMBER PRIMARY KEY,
	    base_Date VARCHAR2(8),
		base_Time VARCHAR2(6),
		category VARCHAR2(3),
		fcst_Date VARCHAR2(8),
		fcst_Time VARCHAR2(4),
		fcst_Value VARCHAR2(12),
		nx NUMBER(5),
		ny NUMBER(5)
	);
	
	CREATE SEQUENCE SEQ_Vilage_Fcst_PK
	START WITH 1
	INCREMENT BY 1
	NOCYCLE;
